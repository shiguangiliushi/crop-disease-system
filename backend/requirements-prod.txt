# WSGI 服务器：runserver 只适合开发（单进程、无超时、有安全提示）
gunicorn>=22.0.0

# 静态文件托管：让 Django 自己把 /static/（admin、DRF 浏览页）发出去。

whitenoise>=6.7.0

# PostgreSQL 驱动（生产建议用 Postgres，别用 sqlite：并发写会锁库）
# 用 psycopg[binary] 免去本机编译依赖。
# psycopg[binary]>=3.2.0

# 共享缓存（第 97 批）：**生产强烈建议装**。

# LocMemCache 是进程内的，gunicorn 起 N 个 worker 就等于把限额放大 N 倍。
# 装好后设 DJANGO_CACHE_URL=redis://127.0.0.1:6379/1 即可启用。
